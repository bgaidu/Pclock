```java
public class BankValidator {
    public static ValidationResult validate(Context ctx, int grade) {
        List<String> missing = new ArrayList<>();
        for (String subject : Subject.ORDER) {
            String path = getBankPath(subject, grade);
            if (!isValidJson(ctx, path)) {
                missing.add(subject);
            }
        }
        return new ValidationResult(missing);
    }
    
    private static boolean isValidJson(Context ctx, String path) {
        try {
            InputStream is = ctx.getAssets().open(path);
            new JsonSlurper().parse(is);
            is.close();
            return true;
        } catch (Exception e) {
            return false;
        }
    }
}
```

**优先级**：P1 | **工作量**：1.5天 | **依赖**：T1

```java
public final class DistractorSampler {
    @Nullable
    public static List<String> sample(@NonNull List<BankEntry> pool,
                                      @NonNull BankEntry chosen, 
                                      boolean answerKey,
                                      @NonNull Random random) {
        // 从同池取 3 个与 chosenVal 不同且互不相同的候选
        Set<String> poolSet = new HashSet<>();
        for (BankEntry entry : pool) {
            poolSet.add(entry.v);
        }
        
        List<String> distractors = new ArrayList<>();
        for (BankEntry entry : pool) {
            if (entry.v.equals(chosen.v)) continue;
            if (distractors.size() >= 3) break;
            if (!distractors.contains(entry.v)) {
                distractors.add(entry.v);
            }
        }
        
        if (distractors.size() < 3) return null;  // 绝不占位
        
        // 拼成4选项并洗牌
        List<String> options = new ArrayList<>();
        options.add(chosen.v);
        options.addAll(distractors);
        
        Collections.shuffle(options, random);
        return options;
    }
}
```

```java
public int getAccuracyPercent() {
    int total = getStatTotal();
    if (total == 0) return -1;  // 无数据

    return (int) Math.round(100.0 * correct / total);
}
```

```java
public final class CustomBankStore {
    private static final String CUSTOM_DIR = "custom_banks";
    
    public static boolean hasCustom(Context ctx, String subject, int grade) {
        File file = new File(ctx.getFilesDir(), CUSTOM_DIR + "/" + subject + "_" + grade + ".json");
        return file.exists();
    }
    
    public static boolean importBank(Context ctx, String subject, int grade, String json) {
        try {
            org.json.JSONObject obj = new org.json.JSONObject(json);
            org.json.JSONArray entries = obj.getJSONArray("entries");
            
            // 校验：条目 ≥ 4，k/v 非空
            if (entries.length() < 4) return false;
            
            // 写入文件
            File dir = new File(ctx.getFilesDir(), CUSTOM_DIR);
            dir.mkdirs();
            File file = new File(dir, subject + "_" + grade + ".json");
            FileUtils.writeStringToFile(file, json, StandardCharsets.UTF_8);
            
            Log.i(TAG, "自定义题库导入成功: " + subject + "_" + grade);
            return true;
        } catch (Exception e) {
            Log.e(TAG, "导入失败", e);
            return false;
        }
    }
    
    @Nullable
    public static BankRepository.LoadResult loadCustom(Context ctx, String subject, int grade) {
        File file = new File(ctx.getFilesDir(), CUSTOM_DIR + "/" + subject + "_" + grade + ".json");
        if (!file.exists()) return null;
        
        String json = FileUtils.readFileToString(file, StandardCharsets.UTF_8);
        org.json.JSONObject obj = new org.json.JSONObject(json);
        org.json.JSONArray entries = obj.getJSONArray("entries");
        
        List<BankEntry> list = new ArrayList<>();
        for (int i = 0; i < entries.length(); i++) {
            org.json.JSONObject entry = entries.getJSONObject(i);
            list.add(new BankEntry(entry.getString("k"), entry.getString("v")));
        }
        
        return new BankRepository.LoadResult(list, 0);  // L0 层级
    }
    
    public static boolean clear(Context ctx, String subject, int grade) {
        File file = new File(ctx.getFilesDir(), CUSTOM_DIR + "/" + subject + "_" + grade + ".json");
        return file.delete();
    }
}
```
